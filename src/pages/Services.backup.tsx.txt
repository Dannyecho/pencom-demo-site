import { useState } from "react";
import { motion } from "framer-motion";
import {
  ArrowRightLeft, UserCircle, TrendingUp, FileCheck, UserPlus,
  Heart, Calculator, Shield, ExternalLink
} from "lucide-react";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import PageBanner from "@/components/PageBanner";
import { services, externalPortals } from "@/lib/data";

const serviceIcons: Record<string, React.ReactNode> = {
  "RSA Transfer": <ArrowRightLeft className="w-6 h-6" />,
  "Personal Pension Plan": <UserPlus className="w-6 h-6" />,
  "Micro Pension": <TrendingUp className="w-6 h-6" />,
  "Pension Calculator": <Calculator className="w-6 h-6" />,
  "Pension Clearance": <FileCheck className="w-6 h-6" />,
  "Online Enrolment": <UserCircle className="w-6 h-6" />,
  "Pencare Health": <Heart className="w-6 h-6" />,
};

export default function Services() {
  const [category, setCategory] = useState<"all" | "individual" | "employer">("all");
  const filtered =
    category === "all" ? services : services.filter((s) => s.category === category);

  return (
    <div className="pb-16">
      <PageBanner
        label="Services"
        title="Pension Services"
        description="Comprehensive pension services for employees, employers, self-employed individuals, and retirees."
        bgImage="/banner-family.jpg"
      />

      <div className="relative z-10 max-w-7xl mx-auto px-4 -mt-8">
        {/* Filter tabs */}
        <div className="flex flex-wrap justify-center gap-3 my-10">
          {(["all", "individual", "employer"] as const).map((cat) => (
            <button
              key={cat}
              onClick={() => setCategory(cat)}
              className={`px-5 py-2.5 rounded-full text-sm font-semibold capitalize transition-all focus-visible:ring-2 focus-visible:ring-[#117F21] focus-visible:ring-offset-2 outline-none ${
                category === cat
                  ? "bg-[#117F21] text-white shadow-md"
                  : "bg-gray-100 text-gray-600 hover:bg-gray-200 active:scale-95"
              }`}
            >
              {cat === "all" ? "All Services" : cat}
            </button>
          ))}
        </div>

        {/* Service cards */}
        <motion.div
          className="grid md:grid-cols-2 lg:grid-cols-3 gap-6"
          initial={{ opacity: 0 }}
          animate={{ opacity: 1 }}
        >
          {filtered.map((service, i) => (
            <motion.div
              key={service.title}
              initial={{ opacity: 0, y: 20 }}
              animate={{ opacity: 1, y: 0 }}
              transition={{ delay: i * 0.08 }}
              className="group bg-white rounded-2xl border border-gray-200/80 p-6 hover:shadow-xl hover:border-[#117F21]/20 hover:-translate-y-1 transition-all duration-300 focus-visible:ring-2 focus-visible:ring-[#117F21] focus-visible:ring-offset-2 outline-none"
            >
              <div className="flex items-start justify-between mb-4">
                <div className="w-12 h-12 rounded-xl bg-[#E8F5E9] flex items-center justify-center text-[#117F21]">
                  {serviceIcons[service.title] || <Shield className="w-6 h-6" />}
                </div>
                <span
                  className={`text-[11px] font-bold uppercase tracking-wider px-2.5 py-1 rounded-full ${
                    service.category === "individual"
                      ? "bg-green-50 text-[#117F21]"
                      : "bg-red-50 text-[#B52026]"
                  }`}
                >
                  {service.category}
                </span>
              </div>
              <h3 className="font-semibold text-gray-900 mb-2 group-hover:text-[#117F21] transition-colors">
                {service.title}
              </h3>
              <p className="text-sm text-gray-500 leading-relaxed">
                {service.description}
              </p>
              <div className="mt-4 flex items-center gap-1 text-sm font-semibold text-[#117F21]">
                <span>Learn more</span>
                <Shield className="w-4 h-4 group-hover:translate-x-1 transition-transform duration-300" />
              </div>
            </motion.div>
          ))}
        </motion.div>

        {/* Pension Calculator */}
        <PensionCalculator />

        {/* External Portals */}
        <div className="mt-16">
          <h2 className="text-2xl font-bold text-[#0A4A13] mb-6 text-center">
            External Portals
          </h2>
          <div className="grid sm:grid-cols-2 lg:grid-cols-4 gap-4">
            {externalPortals.map((portal) => (
              <a
                key={portal.name}
                href={portal.url}
                target="_blank"
                rel="noopener noreferrer"
                className="flex items-center gap-3 p-4 bg-white rounded-xl border border-gray-200/80 hover:border-[#117F21]/30 hover:shadow-md hover:-translate-y-0.5 transition-all duration-300 focus-visible:ring-2 focus-visible:ring-[#117F21] focus-visible:ring-offset-2 outline-none"
              >
                <div className="w-10 h-10 rounded-lg bg-[#E8F5E9] flex items-center justify-center">
                  <ExternalLink className="w-5 h-5 text-[#117F21]" />
                </div>
                <div>
                  <div className="text-sm font-semibold text-gray-800">
                    {portal.name}
                  </div>
                  <div className="text-xs text-gray-400">
                    {portal.description}
                  </div>
                </div>
              </a>
            ))}
          </div>
        </div>
      </div>
    </div>
  );
}

function PensionCalculator() {
  const [salary, setSalary] = useState("");
  const [years, setYears] = useState("");
  const [balance, setBalance] = useState("");
  const [result, setResult] = useState<number | null>(null);

  const calculate = () => {
    const s = Number(salary);
    const y = Number(years);
    const b = Number(balance);
    if (s && y) {
      const employeeContrib = s * 0.08 * 12 * y;
      const employerContrib = s * 0.10 * 12 * y;
      const total = employeeContrib + employerContrib + b;
      setResult(total);
    }
  };

  return (
    <div className="mt-16 bg-white rounded-2xl border border-gray-200 p-6 md:p-8">
      <div className="flex items-center gap-3 mb-6">
        <div className="w-10 h-10 rounded-lg bg-[#CB9615]/10 flex items-center justify-center">
          <Calculator className="w-5 h-5 text-[#CB9615]" />
        </div>
        <div>
          <h2 className="text-xl font-bold text-gray-900">
            Pension Calculator
          </h2>
          <p className="text-sm text-gray-500">
            Estimate your retirement savings
          </p>
        </div>
      </div>
      <div className="grid sm:grid-cols-3 gap-4 mb-6">
        <div>
          <Label htmlFor="salary">Monthly Salary (NGN)</Label>
          <Input
            id="salary"
            type="number"
            value={salary}
            onChange={(e) => setSalary(e.target.value)}
            placeholder="500,000"
            className="mt-1"
          />
        </div>
        <div>
          <Label htmlFor="years">Years of Service</Label>
          <Input
            id="years"
            type="number"
            value={years}
            onChange={(e) => setYears(e.target.value)}
            placeholder="35"
            className="mt-1"
          />
        </div>
        <div>
          <Label htmlFor="balance">Current RSA Balance (NGN)</Label>
          <Input
            id="balance"
            type="number"
            value={balance}
            onChange={(e) => setBalance(e.target.value)}
            placeholder="0"
            className="mt-1"
          />
        </div>
      </div>
      <Button
        onClick={calculate}
        className="bg-[#117F21] hover:bg-[#0A4A13]"
      >
        <Calculator className="w-4 h-4 mr-2" /> Calculate
      </Button>
      {result !== null && (
        <motion.div
          initial={{ opacity: 0, y: 10 }}
          animate={{ opacity: 1, y: 0 }}
          className="mt-6 p-4 bg-[#E8F5E9] rounded-xl"
        >
          <p className="text-sm text-gray-600">
            Estimated Retirement Savings:
          </p>
          <p className="text-3xl font-bold text-[#117F21]">
            N{result.toLocaleString("en-NG")}
          </p>
          <p className="text-xs text-gray-500 mt-1">
            Based on 18% monthly contribution (8% employee + 10% employer)
          </p>
        </motion.div>
      )}
    </div>
  );
}
