/* ─────────────────────────── DONUT CHART COMPONENT ─────────────────────────── */

function DonutChart({ data, size = 180 }: { data: typeof sectorAllocation; size?: number }) {
  const stroke = 22;
  const radius = (size - stroke) / 2;
  const circumference = 2 * Math.PI * radius;
  let offset = 0;
  const total = data.reduce((s, d) => s + d.pct, 0);

  return (
    <div className="relative" style={{ width: size, height: size }}>
      <svg width={size} height={size} className="-rotate-90">
        {data.map((d) => {
          const dash = (d.pct / total) * circumference;
          const seg = (
            <circle
              key={d.sector}
              cx={size / 2}
              cy={size / 2}
              r={radius}
              fill="none"
              stroke={d.color}
              strokeWidth={stroke}
              strokeDasharray={`${dash} ${circumference - dash}`}
              strokeDashoffset={-offset}
              strokeLinecap="butt"
              className="transition-all duration-500 hover:opacity-80 cursor-pointer"
            />
          );
          offset += dash;
          return seg;
        })}
      </svg>
      <div className="absolute inset-0 flex flex-col items-center justify-center">
        <span className="text-xl font-bold text-gray-900">₦20.5T</span>
        <span className="text-[10px] text-gray-400">Total AUM</span>
      </div>
    </div>
  );
}

/* ─────────────────────────── AUM AREA CHART COMPONENT ─────────────────────────── */

function AumAreaChart() {
  const svgRef = useRef<SVGSVGElement>(null);
  const [hoverIdx, setHoverIdx] = useState<number | null>(null);

  const width = 600;
  const height = 200;
  const pad = { top: 20, right: 10, bottom: 30, left: 45 };
  const chartW = width - pad.left - pad.right;
  const chartH = height - pad.top - pad.bottom;

  const minAum = Math.min(...aumHistory.map((d) => d.aum));
  const maxAum = Math.max(...aumHistory.map((d) => d.aum));
  const range = maxAum - minAum;
  const yMin = minAum - range * 0.1;
  const yMax = maxAum + range * 0.15;

  const xScale = (i: number) => pad.left + (i / (aumHistory.length - 1)) * chartW;
  const yScale = (v: number) => pad.top + chartH - ((v - yMin) / (yMax - yMin)) * chartH;

  const pathD = aumHistory
    .map((d, i) => `${i === 0 ? "M" : "L"} ${xScale(i)} ${yScale(d.aum)}`)
    .join(" ");

  const areaD = `${pathD} L ${xScale(aumHistory.length - 1)} ${yScale(yMin)} L ${xScale(0)} ${yScale(yMin)} Z`;

  const yTicks = [yMin, (yMin + yMax) / 2, yMax];

  return (
    <div className="w-full">
      <svg ref={svgRef} viewBox={`0 0 ${width} ${height}`} className="w-full h-auto" preserveAspectRatio="xMidYMid meet">
        <defs>
          <linearGradient id="aumGradient" x1="0" y1="0" x2="0" y2="1">
            <stop offset="0%" stopColor="#117F21" stopOpacity="0.25" />
            <stop offset="100%" stopColor="#117F21" stopOpacity="0.02" />
          </linearGradient>
          <linearGradient id="lineGradient" x1="0" y1="0" x2="1" y2="0">
            <stop offset="0%" stopColor="#2E913D" />
            <stop offset="100%" stopColor="#117F21" />
          </linearGradient>
        </defs>

        {/* Grid lines */}
        {yTicks.map((t, i) => (
          <g key={i}>
            <line x1={pad.left} y1={yScale(t)} x2={width - pad.right} y2={yScale(t)} stroke="#E2E8F0" strokeWidth="0.5" strokeDasharray="3,3" />
            <text x={pad.left - 8} y={yScale(t) + 3} textAnchor="end" fontSize="9" fill="#94A3B8">{t.toFixed(1)}T</text>
          </g>
        ))}

        {/* Area fill */}
        <motion.path
          d={areaD}
          fill="url(#aumGradient)"
          initial={{ opacity: 0 }}
          animate={{ opacity: 1 }}
          transition={{ duration: 1, delay: 0.3 }}
        />

        {/* Line */}
        <motion.path
          d={pathD}
          fill="none"
          stroke="url(#lineGradient)"
          strokeWidth="2.5"
          strokeLinecap="round"
          strokeLinejoin="round"
          initial={{ pathLength: 0 }}
          animate={{ pathLength: 1 }}
          transition={{ duration: 1.5, ease: "easeInOut", delay: 0.2 }}
        />

        {/* Data points */}
        {aumHistory.map((d, i) => (
          <g key={d.month} onMouseEnter={() => setHoverIdx(i)} onMouseLeave={() => setHoverIdx(null)}>
            <motion.circle
              cx={xScale(i)}
              cy={yScale(d.aum)}
              r={hoverIdx === i ? 5 : 3.5}
              fill={hoverIdx === i ? "#117F21" : "#fff"}
              stroke="#117F21"
              strokeWidth="2"
              initial={{ opacity: 0, scale: 0 }}
              animate={{ opacity: 1, scale: 1 }}
              transition={{ delay: 0.8 + i * 0.06 }}
              className="cursor-pointer transition-all"
            />
            {hoverIdx === i && (
              <g>
                <rect x={xScale(i) - 30} y={yScale(d.aum) - 28} width="60" height="18" rx="4" fill="#1E293B" />
                <text x={xScale(i)} y={yScale(d.aum) - 16} textAnchor="middle" fontSize="9" fill="#fff" fontWeight="600">
                  ₦{d.aum}T
                </text>
              </g>
            )}
          </g>
        ))}

        {/* X labels */}
        {aumHistory.map((d, i) => (
          <text key={`x-${i}`} x={xScale(i)} y={height - 8} textAnchor="middle" fontSize="9" fill={hoverIdx === i ? "#117F21" : "#94A3B8"} fontWeight={hoverIdx === i ? 600 : 400}>
            {d.month}
          </text>
        ))}
      </svg>
    </div>
  );
}

            <div className="flex items-center gap-2">
              <div className="w-8 h-8 rounded-lg bg-gradient-to-br from-[#CB9615]/20 to-[#CB9615]/5 flex items-center justify-center">
                <Award className="w-4 h-4 text-[#CB9615]" />
              </div>
              <div>
                <h3 className="font-semibold text-gray-900 text-sm">PFA Performance Rankings</h3>
                <p className="text-[10px] text-gray-400">Ranked by Assets Under Management · May 2026</p>
              </div>
            </div>
            <div className="flex gap-2">
              <button
                onClick={() => setSortField("aum")}
                className={`px-3 py-1.5 rounded-lg text-xs font-semibold transition-all ${sortField === "aum" ? "bg-[#117F21] text-white shadow-sm" : "bg-gray-100 text-gray-600 hover:bg-gray-200"}`}
              >
                By AUM
              </button>
              <button
                onClick={() => setSortField("return")}
                className={`px-3 py-1.5 rounded-lg text-xs font-semibold transition-all ${sortField === "return" ? "bg-[#117F21] text-white shadow-sm" : "bg-gray-100 text-gray-600 hover:bg-gray-200"}`}
              >
                By Return
              </button>
            </div>
          </div>

          {/* Table */}
          <div className="overflow-x-auto">
            <table className="w-full text-sm">
              <thead>
                <tr className="bg-gradient-to-r from-gray-50 to-gray-100/50 text-gray-500">
                  <th className="text-left px-4 py-3 font-semibold text-[10px] uppercase tracking-wider">Rank</th>
                  <th className="text-left px-4 py-3 font-semibold text-[10px] uppercase tracking-wider">PFA Name</th>
                  <th className="text-left px-4 py-3 font-semibold text-[10px] uppercase tracking-wider">AUM</th>
                  <th className="text-left px-4 py-3 font-semibold text-[10px] uppercase tracking-wider">YTD Return</th>
                  <th className="text-left px-4 py-3 font-semibold text-[10px] uppercase tracking-wider">Contributors</th>
                  <th className="text-left px-4 py-3 font-semibold text-[10px] uppercase tracking-wider">Trend</th>
                  <th className="text-left px-4 py-3 font-semibold text-[10px] uppercase tracking-wider">Details</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-gray-50">
                {sortedPfas.map((pfa) => (
                  <>
                    <tr
                      key={pfa.rank}
                      className="hover:bg-[#117F21]/[0.02] transition-colors cursor-pointer group"
                      onClick={() => toggleRow(pfa.rank)}
                    >
                      <td className="px-4 py-3.5">
                        <span className={`w-7 h-7 rounded-full flex items-center justify-center text-xs font-bold shadow-sm ${rankMedal(pfa.rank)}`}>
                          {pfa.rank}
                        </span>
                      </td>
                      <td className="px-4 py-3.5">
                        <span className="font-semibold text-gray-900 text-xs group-hover:text-[#117F21] transition-colors">{pfa.name}</span>
                      </td>
                      <td className="px-4 py-3.5">
                        <span className="font-bold text-gray-900 text-xs">{pfa.aum}</span>
                      </td>
                      <td className="px-4 py-3.5">
                        <span className="inline-flex items-center px-2 py-0.5 rounded-full text-xs font-bold bg-green-50 text-green-700 border border-green-200">
                          {pfa.returnYTD}
                        </span>
                      </td>
                      <td className="px-4 py-3.5 text-gray-500 text-xs">{pfa.contributors}</td>
                      <td className="px-4 py-3.5">
                        {pfa.trend === "up" ? (
                          <span className="flex items-center gap-1 text-xs font-semibold text-green-600">
                            <TrendingUp className="w-3.5 h-3.5" /> Up
                          </span>
                        ) : pfa.trend === "down" ? (
                          <span className="flex items-center gap-1 text-xs font-semibold text-red-600">
                            <TrendingDown className="w-3.5 h-3.5" /> Down
                          </span>
                        ) : (
                          <span className="flex items-center gap-1 text-xs font-semibold text-gray-400">
                            <Minus className="w-3.5 h-3.5" /> Flat
                          </span>
                        )}
                      </td>
                      <td className="px-4 py-3.5">
                        {expandedRows.has(pfa.rank) ? (
                          <ChevronUp className="w-4 h-4 text-[#117F21]" />
                        ) : (
                          <ChevronDown className="w-4 h-4 text-gray-400 group-hover:text-[#117F21] transition-colors" />
                        )}
                      </td>
                    </tr>
                    {expandedRows.has(pfa.rank) && (
                      <tr className="bg-gradient-to-r from-gray-50/80 to-gray-50/40">
                        <td colSpan={7} className="px-4 py-4">
                          <div className="grid grid-cols-3 gap-6">
                            <div className="text-center p-3 bg-white rounded-lg border border-gray-100">
                              <p className="text-[10px] text-gray-400 uppercase tracking-wide">1-Year Return</p>
                              <p className="text-base font-bold text-green-600 mt-1">+{pfa.oneYear}%</p>
                            </div>
                            <div className="text-center p-3 bg-white rounded-lg border border-gray-100">
                              <p className="text-[10px] text-gray-400 uppercase tracking-wide">Market Share</p>
                              <p className="text-base font-bold text-gray-800 mt-1">{pfa.mktShare}%</p>
                            </div>
                            <div className="text-center p-3 bg-white rounded-lg border border-gray-100">
                              <p className="text-[10px] text-gray-400 uppercase tracking-wide">Avg. Monthly Inflow</p>
                              <p className="text-base font-bold text-gray-800 mt-1">{pfa.avgInflow}</p>
                            </div>
                          </div>
                        </td>
                      </tr>
                    )}
                  </>
                ))}
              </tbody>
            </table>
          </div>
        </motion.div>

        {/* ─── Bottom Quick Stats ─── */}
        <motion.div
          initial={{ opacity: 0, y: 20 }}
          animate={{ opacity: 1, y: 0 }}
          transition={{ duration: 0.5, delay: 0.7 }}
          className="grid grid-cols-2 md:grid-cols-4 gap-3"
        >
          {quickStats.map((stat, i) => (
            <motion.div
              key={stat.label}
              initial={{ opacity: 0, y: 15 }}
              animate={{ opacity: 1, y: 0 }}
              transition={{ delay: 0.75 + i * 0.08 }}
              whileHover={{ y: -3 }}
              className="bg-white rounded-xl border border-gray-200 p-4 flex items-center gap-3 shadow-sm hover:shadow-md transition-all cursor-pointer"
            >
              <div className={`w-11 h-11 rounded-xl flex items-center justify-center shadow-sm ${stat.bg}`}>
                <stat.icon className="w-5 h-5 text-white" />
              </div>
              <div>
                <p className="text-lg font-bold text-gray-900">{stat.value}</p>
                <p className="text-[10px] text-gray-500 font-medium">{stat.label}</p>
                <p className="text-[9px] text-gray-400">{stat.sub}</p>
              </div>
            </motion.div>
          ))}
        </motion.div>

        {/* ─── Footer Note ─── */}
        <div className="text-center py-4">
          <p className="text-[10px] text-gray-400">
            Data sourced from PenCom regulatory returns · Updated monthly · Last updated: May 23, 2026
          </p>
        </div>
      </div>
    </div>
  );
