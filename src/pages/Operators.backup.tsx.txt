import { useState } from "react";
import { motion } from "framer-motion";
import {
  Search, Landmark, Phone, MapPin, Users, Shield,
  ChevronDown, ChevronUp, Building2
} from "lucide-react";
import { Input } from "@/components/ui/input";
import PageBanner from "@/components/PageBanner";
import { operators } from "@/lib/data";

const pfcData = [
  { name: "First Pension Custodian Nigeria Limited", address: "Victoria Island, Lagos", phone: "01-2774880", type: "PFC" },
  { name: "UBA Pensions Custodian Limited", address: "Marina, Lagos", phone: "01-2808222", type: "PFC" },
  { name: "Zenith Pensions Custodian Limited", address: "Ajose Adeogun, Victoria Island, Lagos", phone: "01-2787000", type: "PFC" },
  { name: "Sigma Pensions Limited", address: "Ikeja, Lagos", phone: "01-2715000", type: "PFC" },
];

const cpfaData = [
  { name: "APT Pensions Limited", address: "Ikeja, Lagos", phone: "01-4482345", type: "CPFA" },
  { name: "Crusader Pensions Limited", address: "Victoria Island, Lagos", phone: "01-2775700", type: "CPFA" },
  { name: "Fidelity Pensions Limited", address: "Ikeja, Lagos", phone: "01-2778123", type: "CPFA" },
  { name: "IEI-Anchor Pensions Limited", address: "Surulere, Lagos", phone: "01-7744567", type: "CPFA" },
  { name: "NLPC Pensions Fund Administrators Limited", address: "Ikoyi, Lagos", phone: "01-4607890", type: "CPFA" },
  { name: "RAHAMANIYYA Pensions Limited", address: "Abuja", phone: "09-4610000", type: "CPFA" },
];

export default function Operators() {
  const [activeCategory, setActiveCategory] = useState<"pfa" | "pfc" | "cpfa" | "related">("pfa");
  const [searchQuery, setSearchQuery] = useState("");
  const [expandedPfa, setExpandedPfa] = useState<string | null>(null);

  const categories = [
    { key: "pfa" as const, label: "PFAs", count: 22, icon: Landmark },
    { key: "pfc" as const, label: "PFCs", count: 4, icon: Building2 },
    { key: "cpfa" as const, label: "CPFAs", count: 6, icon: Users },
    { key: "related" as const, label: "Related Bodies", count: 2, icon: Shield },
  ];

  const getFilteredData = () => {
    let data;
    switch (activeCategory) {
      case "pfa": data = operators; break;
      case "pfc": data = pfcData; break;
      case "cpfa": data = cpfaData; break;
      case "related":
        data = [
          { name: "Pension Operators Association of Nigeria (PENOp)", address: "Ikoyi, Lagos", phone: "01-2770000", type: "PENOp" },
          { name: "Pension Transitional Arrangement Directorate (PTAD)", address: "Wuse, Abuja", phone: "09-4610000", type: "PTAD" },
        ];
        break;
    }
    if (searchQuery) {
      const q = searchQuery.toLowerCase();
      return data.filter((o) =>
        o.name.toLowerCase().includes(q) || o.address.toLowerCase().includes(q)
      );
    }
    return data;
  };

  const filteredData = getFilteredData();

  return (
    <div className="pb-16">
      <PageBanner
        label="Pension Operators"
        title="Licensed Operators"
        description="Licensed Pension Fund Administrators (PFAs), Pension Fund Custodians (PFCs), and Closed Pension Fund Administrators (CPFAs)."
        bgImage="/banner-elders.jpg"
      />

      <div className="relative z-10 max-w-7xl mx-auto px-4 -mt-8">
        {/* Category Tabs */}
        <div className="flex flex-wrap justify-center gap-3 my-10">
          {categories.map((cat) => (
            <button
              key={cat.key}
              onClick={() => { setActiveCategory(cat.key); setSearchQuery(""); }}
              className={`flex items-center gap-2 px-5 py-2.5 rounded-full text-sm font-medium transition-all ${
                activeCategory === cat.key
                  ? "bg-[#117F21] text-white shadow-md"
                  : "bg-gray-100 text-gray-600 hover:bg-gray-200"
              }`}
            >
              <cat.icon className="w-4 h-4" />
              {cat.label}
              <span className="text-xs opacity-70">({cat.count})</span>
            </button>
          ))}
        </div>

        {/* Search */}
        <div className="max-w-md mx-auto mb-8">
          <div className="relative">
            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
            <Input
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              placeholder={`Search ${activeCategory.toUpperCase()}s by name or city...`}
              className="pl-9 border-gray-200 focus:border-[#117F21] focus:ring-[#117F21]"
            />
          </div>
        </div>

        {/* Table */}
        <motion.div
          className="bg-white rounded-xl border border-gray-200 overflow-hidden"
          initial={{ opacity: 0, y: 20 }}
          animate={{ opacity: 1, y: 0 }}
          transition={{ duration: 0.4 }}
        >
          <div className="overflow-x-auto">
            <table className="w-full text-sm">
              <thead className="bg-gray-50 text-gray-600">
                <tr>
                  <th className="text-left px-5 py-3 font-medium">S/N</th>
                  <th className="text-left px-5 py-3 font-medium">Name</th>
                  <th className="text-left px-5 py-3 font-medium">Location</th>
                  <th className="text-left px-5 py-3 font-medium">Phone</th>
                  {activeCategory === "pfa" && (
                    <th className="text-left px-5 py-3 font-medium">Actions</th>
                  )}
                </tr>
              </thead>
              <tbody className="divide-y divide-gray-100">
                {filteredData.map((op, index) => (
                  <>
                    <motion.tr
                      key={op.name}
                      initial={{ opacity: 0, x: -10 }}
                      animate={{ opacity: 1, x: 0 }}
                      transition={{ delay: index * 0.02 }}
                      className="hover:bg-gray-50/50 transition-colors"
                    >
                      <td className="px-5 py-3 text-gray-500">{index + 1}</td>
                      <td className="px-5 py-3 font-medium text-gray-900">
                        <div className="flex items-center gap-2">
                          <div className="w-8 h-8 rounded-lg bg-[#117F21]/10 flex items-center justify-center shrink-0">
                            <Landmark className="w-4 h-4 text-[#117F21]" />
                          </div>
                          {op.name}
                        </div>
                      </td>
                      <td className="px-5 py-3 text-gray-600">
                        <span className="flex items-center gap-1">
                          <MapPin className="w-3 h-3 text-gray-400" />
                          {op.address}
                        </span>
                      </td>
                      <td className="px-5 py-3 text-gray-600">
                        <span className="flex items-center gap-1">
                          <Phone className="w-3 h-3 text-gray-400" />
                          {op.phone}
                        </span>
                      </td>
                      {activeCategory === "pfa" && (
                        <td className="px-5 py-3">
                          <button
                            onClick={() =>
                              setExpandedPfa(expandedPfa === op.name ? null : op.name)
                            }
                            className="text-[#117F21] text-xs font-medium hover:underline flex items-center gap-1"
                          >
                            {expandedPfa === op.name ? (
                              <>Less <ChevronUp className="w-3 h-3" /></>
                            ) : (
                              <>Details <ChevronDown className="w-3 h-3" /></>
                            )}
                          </button>
                        </td>
                      )}
                    </motion.tr>
                    {activeCategory === "pfa" && expandedPfa === op.name && (
                      <tr className="bg-gray-50/50">
                        <td colSpan={5} className="px-5 py-4">
                          <div className="flex flex-wrap gap-4 text-xs">
                            <span className="px-3 py-1.5 bg-[#E8F5E9] text-[#117F21] rounded-full">
                              RSA Opening Available
                            </span>
                            <span className="px-3 py-1.5 bg-[#E8F5E9] text-[#117F21] rounded-full">
                              Transfer Allowed
                            </span>
                            <span className="px-3 py-1.5 bg-[#E8F5E9] text-[#117F21] rounded-full">
                              Micro Pension
                            </span>
                          </div>
                        </td>
                      </tr>
                    )}
                  </>
                ))}
                {filteredData.length === 0 && (
                  <tr>
                    <td
                      colSpan={activeCategory === "pfa" ? 5 : 4}
                      className="text-center py-12 text-gray-500"
                    >
                      No operators found matching your search.
                    </td>
                  </tr>
                )}
              </tbody>
            </table>
          </div>
        </motion.div>

        {/* PENOp / PTAD detail cards for Related Bodies */}
        {activeCategory === "related" && (
          <div className="mt-8 grid md:grid-cols-2 gap-6">
            <div className="bg-white rounded-xl border border-gray-200 p-6">
              <div className="flex items-center gap-3 mb-4">
                <div className="w-10 h-10 rounded-lg bg-[#1E88E5]/10 flex items-center justify-center">
                  <Users className="w-5 h-5 text-[#1E88E5]" />
                </div>
                <div>
                  <h3 className="font-semibold text-gray-900">PENOp</h3>
                  <p className="text-xs text-gray-500">Pension Operators Association of Nigeria</p>
                </div>
              </div>
              <p className="text-sm text-gray-600 leading-relaxed">
                The industry association representing all licensed pension operators in Nigeria. PENOp advocates for policy reforms, industry best practices, and professional standards across the pension sector.
              </p>
            </div>
            <div className="bg-white rounded-xl border border-gray-200 p-6">
              <div className="flex items-center gap-3 mb-4">
                <div className="w-10 h-10 rounded-lg bg-[#B52026]/10 flex items-center justify-center">
                  <Shield className="w-5 h-5 text-[#B52026]" />
                </div>
                <div>
                  <h3 className="font-semibold text-gray-900">PTAD</h3>
                  <p className="text-xs text-gray-500">Pension Transitional Arrangement Directorate</p>
                </div>
              </div>
              <p className="text-sm text-gray-600 leading-relaxed">
                PTAD manages the pension affairs of retirees under the Defined Benefits Scheme (pre-2004). The directorate ensures timely payment of pension entitlements to federal government retirees.
              </p>
            </div>
          </div>
        )}
      </div>
    </div>
  );
}
